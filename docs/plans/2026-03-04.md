```rust
fn is_system_critical(path_upper: &str) -> bool {
    let critical_paths = [
        "C:\\WINDOWS\\SYSTEM32",
        "C:\\WINDOWS\\SYSWOW64",
        "C:\\WINDOWS\\WINSXS",
        "C:\\PROGRAM FILES\\WINDOWSAPPS",
        "C:\\PROGRAMDATA\\MICROSOFT\\WINDOWS",
        "C:\\SYSTEM VOLUME INFORMATION",
        "C:\\$RECYCLE.BIN",
    ];
    critical_paths.iter().any(|&critical| path_upper.starts_with(critical))
}
```

```rust
fn is_process_running(path: &Path) -> bool {
    let mut sys = System::new_all();
    sys.refresh_all();
    
    for (_pid, process) in sys.processes() {
        if let Some(exe) = process.exe() {
            if exe.starts_with(path) {
                return true;
            }
        }
    }
    false
}
```

```rust
pub fn analyze_migration_safety(path: &Path, size: u64) -> Result<MigrationSafety> {
    let mut score: f32 = 70.0;  // 基础分
    
    // 系统关键路径：直接返回危险
    if is_system_critical(&path_str) {
        return Ok(MigrationSafety {
            risk_level: RiskLevel::Dangerous,
            safety_score: 0,
            can_migrate: false,
            // ...
        });
    }
    
    // 应用类型加分
    match app_type.as_str() {
        "Steam游戏" => score += 20.0,
        "便携式应用" => score += 25.0,
        "用户数据" => score += 30.0,
        // ...
    }
    
    // 运行时检测扣分
    if is_process_running(path) {
        score -= 35.0;
    }
    if is_file_locked(path) {
        score -= 30.0;
    }
    
    // 注册表依赖扣分
    let registry_level = check_registry_dependency(path);
    score -= registry_level as f32 * 10.0;
    
    // 大文件加分
    if size > 10 * 1024 * 1024 * 1024 {
        score += 8.0;
    }
    
    // 限制范围 0-100
    score = score.clamp(0.0, 100.0);
    
    // 确定风险等级
    let risk_level = if score < 30 {
        RiskLevel::Dangerous
    } else if score < 50 {
        RiskLevel::Risky
    } else if score < 75 {
        RiskLevel::Moderate
    } else {
        RiskLevel::Safe
    };
    
    Ok(MigrationSafety {
        risk_level,
        safety_score: score as u8,
        can_migrate: risk_level != RiskLevel::Dangerous,
        reasons,
        recommendations,
        app_type,
    })
}
```

### Task 5: 修复深度扫描后风险提示消失的 bug

#### 问题描述
用户反馈：扫描开始时有风险提示（🟢🟡🟠🔴），但深度扫描完成后，风险提示就消失了。

#### 问题分析
通过代码审查发现：
1. **快速扫描**（`scan_quick`）在第 293 行调用了安全检测：
   ```rust
   let safety = crate::safety::analyze_migration_safety(&path, dir_size).ok();
   ```

2. **深度扫描**（`scan_deep_blocking`）在第 617 行只是设置了 `safety: None`：
   ```rust
   safety: None, // 深度扫描时先不分析，后面统一分析
   ```
   但是后面并没有"统一分析"的代码！

3. 当深度扫描完成后，`deepScanResult` 替换了 `scanResult`，导致所有的安全信息丢失。

#### 解决方案
在深度扫描完成后，为所有目录添加安全分析：

**1. 添加递归分析函数**
```rust
/// 递归分析目录树的安全性
fn analyze_directory_safety(dir: &mut DirectoryNode) {
    use std::path::Path;
    
    // 分析当前目录的安全性
    let path = Path::new(&dir.path);
    dir.safety = crate::safety::analyze_migration_safety(path, dir.size).ok();
    
    // 递归分析所有子目录
    for child in &mut dir.children {
        Self::analyze_directory_safety(child);
    }
}
```

**2. 在构建结果前调用分析**
```rust
let mut directories = root_children;

// 为所有目录添加安全分析
for dir in &mut directories {
    Self::analyze_directory_safety(dir);
}

directories.sort_by(|a, b| b.size.cmp(&a.size));
```

#### 修改文件
- `cdrive-cleaner/src-tauri/src/scanner/disk_scanner.rs`
  - 添加 `analyze_directory_safety` 函数（第 434-447 行）
  - 在 `scan_deep_blocking` 中调用分析（第 691-694 行）

#### 测试验证
```bash
cargo check  # ✅ 编译通过，无错误
```

#### 技术亮点
1. **递归分析**：遍历整个目录树，为每个节点添加安全信息
2. **性能优化**：在构建结果前统一分析，避免重复计算
3. **代码复用**：使用现有的 `analyze_migration_safety` 函数

#### 预期效果
- ✅ 快速扫描时显示风险提示
- ✅ 深度扫描完成后，风险提示依然存在
- ✅ 所有子目录都有完整的安全信息
